${timeStatus}
${iconType}

`; container.appendChild(card); }); } function setPeriod(id, period, el) { const item = rawItems.find(i => i.id === id); if (!item || !item.rates) return; const card = el.closest('.card'); card.querySelectorAll('.period-btn').forEach(b => b.classList.remove('active')); el.classList.add('active'); document.getElementById(`cash-${id}`).textContent = item.rates[period]?.cash || '—'; document.getElementById(`azp-${id}`).textContent = item.rates[period]?.azp || '—'; } function setFilter(cat, el) { activeFilter = cat; document.querySelectorAll('.tab').forEach(b => b.classList.remove('active')); el.classList.add('active'); render(); } document.addEventListener('DOMContentLoaded', init);
